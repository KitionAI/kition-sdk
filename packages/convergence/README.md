# @kition/convergence

The convergence suite of the Kition kit platform for TypeScript engines. A kit whose engine runs as
a `ts-worker` runs it to claim the collaboration level. It has the same scenarios, pass and skip
rules, and report as the Rust `kition-convergence` crate. The suite drives the kit through a small
adapter; `KitEngineAdapter` does the replica part for engines made with `@kition/kit-engine`.
An ES module with type declarations; `@kition/convergence/node` adds the Node-only
`saveReportIfRequested`.

```sh
npm i @kition/convergence
```

```ts
import { DEFAULT_OPTIONS, KitEngineAdapter, run, type EngineDocument, type Rng } from '@kition/convergence'
import { saveReportIfRequested } from '@kition/convergence/node'

class NotesKit extends KitEngineAdapter<View> {
  constructor() {
    super({ name: 'kit.notes', schema: schemaText, model, documentClass: NotesDocument, fileName: 'List.notes' })
  }
  create() {
    return NotesDocument.open('List.notes', sample) as EngineDocument<View>
  }
  randomEdit(doc: EngineDocument<View>, rng: Rng) {
    this.run(doc, 'note.add', { title: `n${rng.below(1000)}` })
    return 'add'
  }
}

const report = run(new NotesKit(), DEFAULT_OPTIONS)
saveReportIfRequested(report) // writes the JSON to the file KITION_CONVERGENCE_REPORT names
if (!report.level6()) throw new Error(report.render())
```

Documentation: https://github.com/KitionAI/kition-sdk/blob/main/docs/platform/kits/levels.md
